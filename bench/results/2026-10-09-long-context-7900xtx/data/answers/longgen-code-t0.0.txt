"""

This module defines the core data structures used throughout the system:
- Book: Represents a book in the library catalog

import re

# ISBN validation pattern: supports ISBN-10 and ISBN-13 formats
# ISBN-10: 9 digits + 1 check digit (0-9 or X)
# ISBN-13: 13 digits, optionally with hyphens
ISBN_PATTERN = re.compile(
    r'^(?:\d{10}|\d{13}|\d{1,5}-\d{1,7}-\d{1,7}-[\dXx])$'
)

# Minimum and maximum values for loan period
MIN_LOAN_DAYS = 1
MAX_LOAN_DAYS = 365

# Maximum number of books a member can borrow at once
MAX_BOOKS_PER_MEMBER = 10


def validate_isbn(isbn: str) -> bool:
    """
    Validate an ISBN string.

    Supports:
    - ISBN-10: 10 digits (last digit can be 'X' or 'x')
    - ISBN-13: 13 digits
    - Hyphenated ISBN-13: e.g., 978-0-13-468599-1

    Args:
        isbn: The ISBN string to validate.

    Returns:
        True if the ISBN is valid, False otherwise.
    """
    if not isbn or not isinstance(isbn, str):
        return False

    # Remove hyphens for basic length check
    clean_isbn = isbn.replace('-', '')

    # Check length: must be 10 or 13 characters after removing hyphens
    if len(clean_isbn) not in (10, 13):
        return False

    # For ISBN-10: first 9 must be digits, last can be digit or X/x
    if len(clean_isbn) == 10:
        if not clean_isbn[:9].isdigit():
            return False
        if clean_isbn[9] not in '0123456789Xx':
            return False
        return True

    # For ISBN-13: all 13 characters must be digits
    if len(clean_isbn) == 13:
        return clean_isbn.isdigit()

    return False


def validate_date(date_str: str) -> bool:
    """
    Validate a date string in YYYY-MM-DD format.

    Args:
        date_str: The date string to validate.

    Returns:
        True if the date is valid, False otherwise.
    """
    if not date_str or not isinstance(date_str, str):
        return False

    try:
        datetime.strptime(date_str, "%Y-%m-%d")
        return True
    except ValueError:
        return False


def validate_positive_int(value: int) -> bool:
    """
    Validate that a value is a positive integer.

    Args:
        value: The value to validate.

    Returns:
        True if the value is a positive integer, False otherwise.
    """
    return isinstance(value, int) and value > 0

@dataclass
class Book:
    """
    Represents a book in the library catalog.

    Attributes:
        id: Unique identifier for the book (auto-generated by database).
        title: Title of the book (required, non-empty).
        author: Author of the book (required, non-empty).
        isbn: International Standard Book Number (required, validated).
        year: Publication year (required, positive integer).
        available: Whether the book is currently available for borrowing.
        created_at: Timestamp when the book was added to the catalog.
    """

    id: Optional[int] = None
    title: str = ""
    author: str = ""
    isbn: str = ""
    year: int = 0
    available: bool = True
    created_at: Optional[datetime] = None

    def __post_init__(self):
        """
        Validate book data after initialization.

        Raises:
            ValueError: If any field fails validation.
        """
        # Validate title
        if not self.title or not isinstance(self.title, str) or not self.title.strip():
            raise ValueError("Book title must be a non-empty string.")

        # Validate author
        if not self.author or not isinstance(self.author, str) or not self.author.strip():
            raise ValueError("Book author must be a non-empty string.")

        # Validate ISBN
        if not validate_isbn(self.isbn):
            raise ValueError(
                f"Invalid ISBN: '{self.isbn}'. "
                "Must be a valid ISBN-10 or ISBN-13."
            )

        # Validate year
        if not isinstance(self.year, int) or self.year <= 0:
            raise ValueError("Book year must be a positive integer.")

        # Validate available flag
        if not isinstance(self.available, bool):
            raise ValueError("Book 'available' must be a boolean.")

        # Normalize title and author (strip whitespace)
        self.title = self.title.strip()
        self.author = self.author.strip()

    def to_dict(self) -> dict:
        """
        Convert the Book instance to a dictionary for serialization.

        Returns:
            A dictionary representation of the book.
        """
        return {
            "id": self.id,
            "title": self.title,
            "author": self.author,
            "isbn": self.isbn,
            "year": self.year,
            "available": self.available,
            "created_at": self.created_at.isoformat() if self.created_at else None,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Book":
        """
        Create a Book instance from a dictionary.

        Args:
            data: Dictionary containing book data.

        Returns:
            A new Book instance.
        """
        created_at = None
        if data.get("created_at"):
            created_at = datetime.fromisoformat(data["created_at"])

        return cls(
            id=data.get("id"),
            title=data.get("title", ""),
            author=data.get("author", ""),
            isbn=data.get("isbn", ""),
            year=data.get("year", 0),
            available=data.get("available", True),
            created_at=created_at,
        )

    def __str__(self) -> str:
        """Return a human-readable string representation of the book."""
        status = "Available" if self.available else "Checked Out"
        return (
            f"[{self.id}] \"{self.title}\" by {self.author} "
            f"(ISBN: {self.isbn}, Year: {self.year}) - {status}"
        )

@dataclass
class Member:
    """
    Represents a library member.

    Attributes:
        id: Unique identifier for the member (auto-generated by database).
        name: Full name of the member (required, non-empty).
        email: Email address of the member (required, validated).
        joined_at: Date when the member joined the library.
        active: Whether the member's account is active.
    """

    id: Optional[int] = None
    name: str = ""
    email: str = ""
    joined_at: Optional[date] = None
    active: bool = True

    def __post_init__(self):
        """
        Validate member data after initialization.

        Raises:
            ValueError: If any field fails validation.
        """
        # Validate name
        if not self.name or not isinstance(self.name, str) or not self.name.strip():
            raise ValueError("Member name must be a non-empty string.")

        # Validate email
        if not self.email or not isinstance(self.email, str):
            raise ValueError("Member email must be a non-empty string.")

        # Basic email validation: must contain '@' and a domain
        if "@" not in self.email:
            raise ValueError(f"Invalid email address: '{self.email}'.")

        # Validate joined_at date
        if self.joined_at is not None:
            if not isinstance(self.joined_at, date):
                raise ValueError("Member joined_at must be a date object.")

        # Validate active flag
        if not isinstance(self.active, bool):
            raise ValueError("Member 'active' must be a boolean.")

        # Normalize name and email
        self.name = self.name.strip()
        self.email = self.email.strip().lower()

    def to_dict(self) -> dict:
        """
        Convert the Member instance to a dictionary for serialization.

        Returns:
            A dictionary representation of the member.
        """
        return {
            "id": self.id,
            "name": self.name,
            "email": self.email,
            "joined_at": self.joined_at.isoformat() if self.joined_at else None,
            "active": self.active,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Member":
        """
        Create a Member instance from a dictionary.

        Args:
            data: Dictionary containing member data.

        Returns:
            A new Member instance.
        """
        joined_at = None
        if data.get("joined_at"):
            joined_at = date.fromisoformat(data["joined_at"])

        return cls(
            id=data.get("id"),
            name=data.get("name", ""),
            email=data.get("email", ""),
            joined_at=joined_at,
            active=data.get("active", True),
        )

    def __str__(self) -> str:
        """Return a human-readable string representation of the member."""
        status = "Active" if self.active else "Inactive"
        joined = self.joined_at.isoformat() if self.joined_at else "N/A"
        return (
            f"[{self.id}] {self.name} <{self.email}> "
            f"(Joined: {joined}) - {status}"

@dataclass
class Loan:
    """
    Represents a loan transaction between a member and a book.

    Attributes:
        id: Unique identifier for the loan (auto-generated by database).
        book_id: ID of the borrowed book.
        member_id: ID of the borrowing member.
        loan_date: Date when the book was borrowed.
        due_date: Date when the book is due back.
        return_date: Date when the book was returned (None if not yet returned).
        status: Current status of the loan ('active', 'returned', 'overdue').
    """

    id: Optional[int] = None
    book_id: int = 0
    member_id: int = 0
    loan_date: Optional[date] = None
    due_date: Optional[date] = None
    return_date: Optional[date] = None
    status: str = "active"

    def __post_init__(self):
        """
        Validate loan data after initialization.

        Raises:
            ValueError: If any field fails validation.
        """
        # Validate book_id
        if not validate_positive_int(self.book_id):
            raise ValueError("Loan book_id must be a positive integer.")

        # Validate member_id
        if not validate_positive_int(self.member_id):
            raise ValueError("Loan member_id must be a positive integer.")

        # Validate loan_date
        if self.loan_date is None:
            raise ValueError("Loan loan_date is required.")
        if not isinstance(self.loan_date, date):
            raise ValueError("Loan loan_date must be a date object.")

        # Validate due_date
        if self.due_date is None:
            raise ValueError("Loan due_date is required.")
        if not isinstance(self.due_date, date):
            raise ValueError("Loan due_date must be a date object.")

        # Due date must be after loan date
        if self.due_date < self.loan_date:
            raise ValueError("Loan due_date must be after loan_date.")

        # Validate return_date if present
        if self.return_date is not None:
            if not isinstance(self.return_date, date):
                raise ValueError("Loan return_date must be a date object.")
            if self.return_date < self.loan_date:
                raise ValueError("Loan return_date must be on or after loan_date.")

        # Validate status
        valid_statuses = {"active", "returned", "overdue"}
        if self.status not in valid_statuses:
            raise ValueError(
                f"Invalid loan status: '{self.status}'. "
                f"Must be one of: {valid_statuses}"
            )

    def is_overdue(self, reference_date: Optional[date] = None) -> bool:
        """
        Check if the loan is overdue.

        Args:
            reference_date: The date to check against. Defaults to today.

        Returns:
            True if the loan is overdue, False otherwise.
        """
        if self.status == "returned":
            return False

        if reference_date is None:
            reference_date = date.today()

        return reference_date > self.due_date

    def days_overdue(self, reference_date: Optional[date] = None) -> int:
        """
        Calculate the number of days the loan is overdue.

        Args:
            reference_date: The date to check against. Defaults to today.

        Returns:
            Number of days overdue (0 if not overdue).
        """
        if not self.is_overdue(reference_date):
            return 0

        if reference_date is None:
            reference_date = date.today()

        return (reference_date - self.due_date).days

    def to_dict(self) -> dict:
        """
        Convert the Loan instance to a dictionary for serialization.

        Returns:
            A dictionary representation of the loan.
        """
        return {
            "id": self.id,
            "book_id": self.book_id,
            "member_id": self.member_id,
            "loan_date": self.loan_date.isoformat() if self.loan_date else None,
            "due_date": self.due_date.isoformat() if self.due_date else None,
            "return_date": self.return_date.isoformat() if self.return_date else None,
            "status": self.status,
        }

    @classmethod
    def from_dict(cls, data: dict) -> "Loan":
        """
        Create a Loan instance from a dictionary.

        Args:
            data: Dictionary containing loan data.

        Returns:
            A new Loan instance.
        """
        loan_date = None
        due_date = None
        return_date = None

        if data.get("loan_date"):
            loan_date = date.fromisoformat(data["loan_date"])
        if data.get("due_date"):
            due_date = date.fromisoformat(data["due_date"])
        if data.get("return_date"):
            return_date = date.fromisoformat(data["return_date"])

        return cls(
            id=data.get("id"),
            book_id=data.get("book_id", 0),
            member_id=data.get("member_id", 0),
            loan_date=loan_date,
            due_date=due_date,
            return_date=return_date,
            status=data.get("status", "active"),
        )

    def __str__(self) -> str:
        """Return a human-readable string representation of the loan."""
        loan_str = self.loan_date.isoformat() if self.loan_date else "N/A"
        due_str = self.due_date.isoformat() if self.due_date else "N/A"
        return_str = self.return_date.isoformat() if self.return_date else "Not returned"
        return (
            f"[{self.id}] Book#{self.book_id} -> Member#{self.member_id} "
            f"(Loaned: {loan_str}, Due: {due_str}, Returned: {return_str}) "
            f"- Status: {self.status}"
        )

"""
SQLite-based persistence layer for the Library Management System.

This module provides a LibraryStore class that handles all database operations:
- Creating and managing the SQLite database schema
- CRUD operations for books, members, and loans
- Query operations for searching and filtering
- Transaction management for data consistency

The store uses parameterized queries to prevent SQL injection.
"""

import sqlite3
from datetime import date, datetime
from typing import List, Optional, Tuple
from contextlib import contextmanager

from library.models import Book, Member, Loan


# =============================================================================
# Database Schema
# =============================================================================

# SQL statements to create the database tables
CREATE_TABLES_SQL = """
CREATE TABLE IF NOT EXISTS books (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    title TEXT NOT NULL,
    author TEXT NOT NULL,
    isbn TEXT NOT NULL UNIQUE,
    year INTEGER NOT NULL,
    available INTEGER NOT NULL DEFAULT 1,
    created_at TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS members (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE,
    joined_at TEXT NOT NULL,
    active INTEGER NOT NULL DEFAULT 1
);

CREATE TABLE IF NOT EXISTS loans (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    book_id INTEGER NOT NULL,
    member_id INTEGER NOT NULL,
    loan_date TEXT NOT NULL,
    due_date TEXT NOT NULL,
    return_date TEXT,
    status TEXT NOT NULL DEFAULT 'active',
    FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE,
    FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS idx_books_isbn ON books(isbn);
CREATE INDEX IF NOT EXISTS idx_books_title ON books(title);
CREATE INDEX IF NOT EXISTS idx_members_email ON members(email);
CREATE INDEX IF NOT EXISTS idx_loans_book_id ON loans(book_id);
CREATE INDEX IF NOT EXISTS idx_loans_member_id ON loans(member_id);
CREATE INDEX IF NOT EXISTS idx_loans_status ON loans(status);
"""


# =============================================================================
# LibraryStore Class
# =============================================================================

class LibraryStore:
    """
    SQLite-based storage layer for the library management system.

    This class encapsulates all database interactions, providing a clean
    interface for CRUD operations on books, members, and loans.

    Usage:
        store = LibraryStore("library.db")
        store.initialize()
        book = store.add_book(Book(title="...", author="...", isbn="...", year=2020))
        store.close()
    """

    def __init__(self, db_path: str = ":memory:"):
        """
        Initialize the LibraryStore.

        Args:
            db_path: Path to the SQLite database file.
                     Use ":memory:" for an in-memory database (useful for testing).
        """
        self.db_path = db_path
        self._connection: Optional[sqlite3.Connection] = None

    def initialize(self) -> None:
        """
        Initialize the database connection and create tables if they don't exist.

        This method should be called before any other operations.
        It is safe to call multiple times (uses CREATE TABLE IF NOT EXISTS).
        """
        self._connection = sqlite3.connect(self.db_path)
        # Enable foreign key constraints
        self._connection.execute("PRAGMA foreign_keys = ON;")
        # Create all tables
        self._connection.executescript(CREATE_TABLES_SQL)
        self._connection.commit()

    def close(self) -> None:
        """
        Close the database connection.

        Should be called when the store is no longer needed.
        """
        if self._connection:
            self._connection.close()
            self._connection = None

    @contextmanager
    def transaction(self):
        """
        Context manager for database transactions.

        Ensures that either all operations in the block succeed (commit)
        or none do (rollback), maintaining data consistency.

        Usage:
            with store.transaction():
                store.add_book(book)
                store.add_member(member)
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        try:
            yield self._connection
            self._connection.commit()
        except Exception:
            self._connection.rollback()
            raise

    # -------------------------------------------------------------------------
    # Book Operations
    # -------------------------------------------------------------------------

    def add_book(self, book: Book) -> Book:
        """
        Add a new book to the catalog.

        Args:
            book: The Book instance to add.

        Returns:
            The Book instance with its assigned ID.

        Raises:
            ValueError: If the book data is invalid.
            sqlite3.IntegrityError: If the ISBN already exists.
            RuntimeError: If the database is not initialized.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            """
            INSERT INTO books (title, author, isbn, year, available, created_at)
            VALUES (?, ?, ?, ?, ?, ?)
            """,
            (
                book.title,
                book.author,
                book.isbn,
                book.year,
                1 if book.available else 0,
                datetime.now().isoformat(),
            ),
        )
        self._connection.commit()

        # Return the book with its new ID
        book.id = cursor.lastrowid
        book.created_at = datetime.now()
        return book

    def get_book_by_id(self, book_id: int) -> Optional[Book]:
        """
        Retrieve a book by its ID.

        Args:
            book_id: The ID of the book to retrieve.

        Returns:
            The Book instance if found, None otherwise.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            "SELECT id, title, author, isbn, year, available, created_at FROM books WHERE id = ?",
            (book_id,),
        )
        row = cursor.fetchone()

        if row is None:
            return None

        return self._row_to_book(row)

    def get_book_by_isbn(self, isbn: str) -> Optional[Book]:
        """
        Retrieve a book by its ISBN.

        Args:
            isbn: The ISBN of the book to retrieve.

        Returns:
            The Book instance if found, None otherwise.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            "SELECT id, title, author, isbn, year, available, created_at FROM books WHERE isbn = ?",
            (isbn,),
        )
        row = cursor.fetchone()

        if row is None:
            return None

        return self._row_to_book(row)

    def search_books(self, query: str) -> List[Book]:
        """
        Search books by title or author (case-insensitive partial match).

        Args:
            query: The search term to match against title or author.

        Returns:
            A list of matching Book instances.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        search_pattern = f"%{query}%"
        cursor = self._connection.execute(
            """
            SELECT id, title, author, isbn, year, available, created_at
            FROM books
            WHERE title LIKE ? OR author LIKE ?
            ORDER BY title ASC
            """,
            (search_pattern, search_pattern),
        )
        rows = cursor.fetchall()

        return [self._row_to_book(row) for row in rows]

    def list_all_books(self) -> List[Book]:
        """
        Retrieve all books in the catalog.

        Returns:
            A list of all Book instances.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            "SELECT id, title, author, isbn, year, available, created_at FROM books ORDER BY title ASC"
        )
        rows = cursor.fetchall()

        return [self._row_to_book(row) for row in rows]

    def update_book(self, book: Book) -> bool:
        """
        Update an existing book in the catalog.

        Args:
            book: The Book instance with updated fields. Must have a valid ID.

        Returns:
            True if the book was updated, False if no book with that ID exists.

        Raises:
            ValueError: If the book data is invalid.
            RuntimeError: If the database is not initialized.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        if book.id is None:
            raise ValueError("Book must have an ID to be updated.")

        cursor = self._connection.execute(
            """
            UPDATE books
            SET title = ?, author = ?, isbn = ?, year = ?, available = ?
            WHERE id = ?
            """,
            (
                book.title,
                book.author,
                book.isbn,
                book.year,
                1 if book.available else 0,
                book.id,
            ),
        )
        self._connection.commit()

        return cursor.rowcount > 0

    def delete_book(self, book_id: int) -> bool:
        """
        Delete a book from the catalog.

        Note: This will also delete associated loans due to ON DELETE CASCADE.

        Args:
            book_id: The ID of the book to delete.

        Returns:
            True if the book was deleted, False if no book with that ID exists.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute("DELETE FROM books WHERE id = ?", (book_id,))
        self._connection.commit()

        return cursor.rowcount > 0

    def set_book_availability(self, book_id: int, available: bool) -> bool:
        """
        Set the availability status of a book.

        Args:
            book_id: The ID of the book.
            available: True if the book is available, False otherwise.

        Returns:
            True if the book was updated, False if no book with that ID exists.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            "UPDATE books SET available = ? WHERE id = ?",
            (1 if available else 0, book_id),
        )
        self._connection.commit()

        return cursor.rowcount > 0

    # -------------------------------------------------------------------------
    # Member Operations
    # -------------------------------------------------------------------------

    def add_member(self, member: Member) -> Member:
        """
        Add a new member to the library.

        Args:
            member: The Member instance to add.

        Returns:
            The Member instance with its assigned ID.

        Raises:
            ValueError: If the member data is invalid.
            sqlite3.IntegrityError: If the email already exists.
            RuntimeError: If the database is not initialized.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        joined_at = member.joined_at or date.today()

        cursor = self._connection.execute(
            """
            INSERT INTO members (name, email, joined_at, active)
            VALUES (?, ?, ?, ?)
            """,
            (
                member.name,
                member.email,
                joined_at.isoformat(),
                1 if member.active else 0,
            ),
        )
        self._connection.commit()

        # Return the member with its new ID
        member.id = cursor.lastrowid
        member.joined_at = joined_at
        return member

    def get_member_by_id(self, member_id: int) -> Optional[Member]:
        """
        Retrieve a member by their ID.

        Args:
            member_id: The ID of the member to retrieve.

        Returns:
            The Member instance if found, None otherwise.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        cursor = self._connection.execute(
            "SELECT id, name, email, joined_at, active FROM members WHERE id = ?",
            (member_id,),
        )
        row = cursor.fetchone()

    def create_loan(self, book_id: int, member_id: int, loan_days: int = 14) -> Loan:
        """
        Create a new loan for a book to a member.

        Validates:
        - The book exists and is available.
        - The member exists and is active.
        - The member has not exceeded the maximum number of borrowed books.
        - The loan period is within allowed bounds.

        Args:
            book_id: The ID of the book to borrow.
            member_id: The ID of the borrowing member.
            loan_days: Number of days for the loan (default: 14).

        Returns:
            The newly created Loan instance.

        Raises:
            ValueError: If validation fails.
            RuntimeError: If the database is not initialized.
        """
        if self._connection is None:
            raise RuntimeError("Database not initialized. Call initialize() first.")

        # Validate loan period
        if not validate_positive_int(loan_days):
            raise ValueError("Loan days must be a positive integer.")
        if loan_days < MIN_LOAN_DAYS or loan_days > MAX_LOAN_DAYS:
            raise ValueError(
                f"Loan period must be between {MIN_LOAN_DAYS} and {MAX_LOAN_DAYS} days."
            )

        # Check book exists and is available
        book = self.get_book_by_id(book_id)
        if book is None:
            raise ValueError(f"Book with ID {book_id} not found.")
        if not book.available:
            raise ValueError(f"Book '{book.title}' (ID {book_id}) is not available.")

        # Check member exists and is active
        member = self.get_member_by_id(member_id)
        if member is None:
            raise ValueError(f"Member with ID {member_id} not found.")
        if not member.active:
            raise ValueError(f"Member '{member.name}' (ID {member_id}) is not active.")

        # Check member hasn't exceeded max books
        active_loans = self.get_active_loans_for_member(member_id)
        if len(active_loans) >= MAX_BOOKS_PER_MEMBER:
            raise ValueError(
                f"Member '{member.name}' has already borrowed "
                f"{MAX_BOOKS_PER_MEMBER} books."
            )

        # Create the loan
        loan_date = date.today()
        due_date = loan_date + __import__('datetime').timedelta(days=loan_days)

        cursor = self._connection.execute(
            """
            INSERT INTO loans (book_id, member_id, loan_date, due_date, status)
            VALUES (?, ?, ?, ?, ?)
            """,
            (
                book_id,
                member_id,
                loan_date.isoformat(),
                due_date.isoformat(),
                "active",
            ),
        )
        self._connection.commit()

        # Mark book as unavailable
        self.set_book_availability(book_id, False)

        # Return the loan with its new ID
        loan = Loan(
            id=cursor.lastrowid,
            book_id=book_id,
            member_id=member_id,
            loan_date=loan_date,
            due_date=due_date,
            status="active",
        )
        return loan

    def return_loan(self, loan_id: int) -> Loan:

    Main entry point for the CLI.

    Parses arguments, initializes the store, and dispatches to the
    appropriate command handler.
    """
    parser = build_parser()
    args = parser.parse_args()

        parser.print_help()

    # Initialize the store
    store = LibraryStore(db_path=args.db)
    store.initialize()

    try:
        # Dispatch to the appropriate command handler
        command_handlers = {
            "add-book": cmd_add_book,
            "list-books": cmd_list_books,
            "search-books": cmd_search_books,
            "get-book": cmd_get_book,
            "update-book": cmd_update_book,

            "return": cmd_return,
            "list-loans": cmd_list_loans,
            "overdue": cmd_overdue,
            "stats": cmd_stats,
        }

        handler = command_handlers.get(args.command)
        if handler is None:
            print(f"Unknown command: {args.command}", file=sys.stderr)
            parser.print_help()
            sys.exit(1)

        handler(store, args)

    finally:
        store.close()
