# Library Management System

Below is a complete, well-commented Python library management system split into logical files.

## File 1: `models.py` — Data Model

```python
"""
models.py
---------
Defines the core data classes for the library management system.

Classes:
    Book          – represents a single book in the catalogue
    Member        – represents a library member (patron)
    Loan          – represents a loan (checkout) of a book to a member

Each class uses __repr__ for readable debugging output and __eq__ where
useful.  Dataclasses are used to minimise boilerplate.
"""

from __future__ import annotations

import dataclasses
import datetime
from typing import Optional


@dataclasses.dataclass(frozen=True)
class Book:
    """
    A single book in the library catalogue.

    Attributes
    ----------
    isbn      : str   – International Standard Book Number (10 or 13 digits)
    title     : str   – Full title of the book
    author    : str   – Primary author name
    year      : int   – Publication year
    available : bool  – True when the book is on the shelf and can be borrowed
    """
    isbn: str
    title: str
    author: str
    year: int
    available: bool = True

    def __post_init__(self) -> None:
        """Validate the book data at construction time."""
        # ISBN must be 10 or 13 digits (allow dashes for readability)
        clean_isbn = self.isbn.replace("-", "")
        if not clean_isbn.isdigit():
            raise ValueError(f"ISBN must contain only digits: {self.isbn!r}")
        if len(clean_isbn) not in (10, 13):
            raise ValueError(
                f"ISBN must be 10 or 13 digits (got {len(clean_isbn)}): {self.isbn!r}"
            )

        # Title and author must be non-empty strings
        if not self.title or not self.title.strip():
            raise ValueError("Title must be a non-empty string")
        if not self.author or not self.author.strip():
            raise ValueError("Author must be a non-empty string")

        # Year must be a reasonable publication year
        if self.year < 1000 or self.year > datetime.date.today().year:
            raise ValueError(
                f"Year must be between 1000 and {datetime.date.today().year}: {self.year}"
            )

    def __repr__(self) -> str:
        status = "Available" if self.available else "Checked out"
        return (
            f"Book(isbn={self.isbn!r}, title={self.title!r}, "
            f"author={self.author!r}, year={self.year}, status={status})"
        )


@dataclasses.dataclass(frozen=True)
class Member:
    """
    A library member (patron).

    Attributes
    ----------
    member_id : int  – Unique numeric identifier
    name      : str  – Full name of the member
    email     : str  – Email address
    """
    member_id: int
    name: str
    email: str

    def __post_init__(self) -> None:
        """Validate member data at construction time."""
        if self.member_id <= 0:
            raise ValueError(f"Member ID must be positive: {self.member_id}")
        if not self.name or not self.name.strip():
            raise ValueError("Member name must be a non-empty string")
        # Very simple email validation – just check for '@'
        if "@" not in self.email:
            raise ValueError(f"Invalid email address: {self.email!r}")

    def __repr__(self) -> str:
        return (
            f"Member(member_id={self.member_id}, name={self.name!r}, "
            f"email={self.email!r})"
        )


@dataclasses.dataclass(frozen=True)
class Loan:
    """
    A loan record: a book checked out by a member on a given date.

    Attributes
    ----------
    loan_id       : int            – Unique numeric identifier
    book_isbn     : str            – ISBN of the borrowed book
    member_id     : int            – ID of the borrowing member
    due_date      : datetime.date  – Date by which the book must be returned
    returned_date : Optional[datetime.date] – Date the book was actually returned (None if still out)
    """
    loan_id: int
    book_isbn: str
    member_id: int
    due_date: datetime.date
    returned_date: Optional[datetime.date] = None

    def __post_init__(self) -> None:
        """Validate loan data at construction time."""
        if self.loan_id <= 0:
            raise ValueError(f"Loan ID must be positive: {self.loan_id}")
        if self.member_id <= 0:
            raise ValueError(f"Member ID must be positive: {self.member_id}")
        # ISBN basic check
        clean_isbn = self.book_isbn.replace("-", "")
        if not clean_isbn.isdigit():
            raise ValueError(f"Book ISBN must contain only digits: {self.book_isbn!r}")
        if self.due_date < datetime.date(1900, 1, 1):
            raise ValueError(f"Due date is too early: {self.due_date}")
        if self.returned_date is not None and self.returned_date < self.due_date:
            # Returning before the due date is fine, but we flag illogical dates
            pass  # early return is acceptable

    def is_active(self) -> bool:
        """Return True if the loan is still open (book not yet returned)."""
        return self.returned_date is None

    def is_overdue(self, as_of: Optional[datetime.date] = None) -> bool:
        """Return True if the loan is still open and past its due date."""
        if not self.is_active():
            return False
        today = as_of or datetime.date.today()
        return today > self.due_date

    def __repr__(self) -> str:
        status = "Returned" if not self.is_active() else "Active"
        overdue_tag = " [OVERDUE]" if self.is_overdue() else ""
        return (
            f"Loan(loan_id={self.loan_id}, isbn={self.book_isbn!r}, "
            f"member_id={self.member_id}, due={self.due_date}, "
            f"status={status}{overdue_tag})"
        )
```

## File 2: `store.py` — SQLite Storage Layer

```python
"""
store.py
--------
SQLite-backed persistence layer for the library management system.

The LibraryStore class owns a single SQLite connection (in-memory by default)
and exposes high-level methods that return fully-populated model objects
(Book, Member, Loan).  All queries use parameterised statements to prevent
SQL injection.

Schema tables
-------------
books   : isbn TEXT PRIMARY KEY, title TEXT, author TEXT, year INTEGER, available INTEGER (0/1)
members : member_id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT, email TEXT
loans   : loan_id INTEGER PRIMARY KEY AUTOINCREMENT, book_isbn TEXT, member_id INTEGER,
          due_date TEXT (ISO), returned_date TEXT (ISO or NULL)
"""

from __future__ import annotations

import datetime
import sqlite3
from typing import List, Optional

from models import Book, Member, Loan


class LibraryStore:
    """
    Thin wrapper around a SQLite database for the library catalogue.

    Parameters
    ----------
    db_path : str | None
        Path to the SQLite file.  Use ``":memory:"`` (default) for a
        throw-away database (ideal for tests).  Use a real path for
        persistence.
    """

    # ── Schema DDL ──────────────────────────────────────────────
    _SCHEMA = """
    CREATE TABLE IF NOT EXISTS books (
        isbn      TEXT    PRIMARY KEY,
        title     TEXT    NOT NULL,
        author    TEXT    NOT NULL,
        year      INTEGER NOT NULL,
        available INTEGER NOT NULL DEFAULT 1
    );

    CREATE TABLE IF NOT EXISTS members (
        member_id INTEGER PRIMARY KEY AUTOINCREMENT,
        name      TEXT    NOT NULL,
        email     TEXT    NOT NULL
    );

    CREATE TABLE IF NOT EXISTS loans (
        loan_id       INTEGER PRIMARY KEY AUTOINCREMENT,
        book_isbn     TEXT    NOT NULL,
        member_id     INTEGER NOT NULL,
        due_date      TEXT    NOT NULL,
        returned_date TEXT    NULL,
        FOREIGN KEY (book_isbn) REFERENCES books(isbn),
        FOREIGN KEY (member_id) REFERENCES members(member_id)
    );
    """

    def __init__(self, db_path: str = ":memory:") -> None:
        self._conn = sqlite3.connect(db_path)
        # Return rows as tuple-like objects so we can access by column name
        self._conn.row_factory = sqlite3.Row
        # Enforce foreign-key constraints
        self._conn.execute("PRAGMA foreign_keys = ON;")
        # Create tables if they don't already exist
        self._conn.executescript(self._SCHEMA)
        self._conn.commit()

    # ── Lifecycle ───────────────────────────────────────────────
    def close(self) -> None:
        """Close the underlying SQLite connection."""
        self._conn.close()

    # ── Books ───────────────────────────────────────────────────
    def add_book(self, book: Book) -> Book:
        """
        Insert a new book into the catalogue.

        Raises
        ------
        sqlite3.IntegrityError
            If a book with the same ISBN already exists.
        """
        self._conn.execute(
            "INSERT INTO books (isbn, title, author, year, available) "
            "VALUES (?, ?, ?, ?, ?)",
            (book.isbn, book.title, book.author, book.year, int(book.available)),
        )
        self._conn.commit()
        return book

    def get_book(self, isbn: str) -> Optional[Book]:
        """Fetch a single book by ISBN.  Returns None if not found."""
        row = self._conn.execute(
            "SELECT * FROM books WHERE isbn = ?", (isbn,)
        ).fetchone()
        if row is None:
            return None
        return self._row_to_book(row)

    def list_books(self) -> List[Book]:
        """Return every book in the catalogue, ordered by title."""
        rows = self._conn.execute(
            "SELECT * FROM books ORDER BY title"
        ).fetchall()
        return [self._row_to_book(r) for r in rows]

    def search_books(self, query: str) -> List[Book]:
        """
        Case-insensitive partial search on title and author.

        Parameters
        ----------
        query : str
            Substring to look for (matches title OR author).
        """
        pattern = f"%{query}%"
        rows = self._conn.execute(
            "SELECT * FROM books WHERE title LIKE ? OR author LIKE ? ORDER BY title",
            (pattern, pattern),
        ).fetchall()
        return [self._row_to_book(r) for r in rows]

    def remove_book(self, isbn: str) -> bool:
        """
        Delete a book by ISBN.  Returns True if a row was deleted,
        False if the ISBN was not found.

        Raises
        ------
        sqlite3.IntegrityError
            If the book is currently on loan (foreign-key constraint).
        """
        cursor = self._conn.execute(
            "DELETE FROM books WHERE isbn = ?", (isbn,)
        )
        self._conn.commit()
        return cursor.rowcount > 0

    # ── Members ─────────────────────────────────────────────────
    def add_member(self, name: str, email: str) -> Member:
        """
        Insert a new member and return the Member with its assigned ID.
        """
        cursor = self._conn.execute(
            "INSERT INTO members (name, email) VALUES (?, ?)",
            (name, email),
        )
        self._conn.commit()
        return Member(member_id=cursor.lastrowid, name=name, email=email)

    def get_member(self, member_id: int) -> Optional[Member]:
        """Fetch a single member by ID.  Returns None if not found."""
        row = self._conn.execute(
            "SELECT * FROM members WHERE member_id = ?", (member_id,)
        ).fetchone()
        if row is None:
            return None
        return self._row_to_member(row)

    def list_members(self) -> List[Member]:
        """Return every member, ordered by name."""
        rows = self._conn.execute(
            "SELECT * FROM members ORDER BY name"
        ).fetchall()
        return [self._row_to_member(r) for r in rows]

    def remove_member(self, member_id: int) -> bool:
        """
        Delete a member by ID.  Returns True if deleted, False otherwise.

        Raises
        ------
        sqlite3.IntegrityError
            If the member has outstanding loans.
        """
        cursor = self._conn.execute(
            "DELETE FROM members WHERE member_id = ?", (member_id,)
        )
        self._conn.commit()
        return cursor.rowcount > 0

    # ── Loans ───────────────────────────────────────────────────
    def create_loan(self, book_isbn: str, member_id: int, due_date: datetime.date) -> Loan:
        """
        Check out a book to a member.

        Steps
        -----
        1. Verify the book exists and is available.
        2. Verify the member exists.
        3. Mark the book as unavailable.
        4. Insert the loan record.

        Raises
        ------
        ValueError  – book not found or already checked out, member not found.
        """
        book = self.get_book(book_isbn)
        if book is None:
            raise ValueError(f"Book with ISBN {book_isbn!r} not found")
        if not book.available:
            raise ValueError(f"Book {book.title!r} is already checked out")

        member = self.get_member(member_id)
        if member is None:
            raise ValueError(f"Member with ID {member_id} not found")

        # Mark book as unavailable
        self._conn.execute(
            "UPDATE books SET available = 0 WHERE isbn = ?", (book_isbn,)
        )

        # Insert loan
        cursor = self._conn.execute(
            "INSERT INTO loans (book_isbn, member_id, due_date) VALUES (?, ?, ?)",
            (book_isbn, member_id, due_date.isoformat()),
        )
        self._conn.commit()

        return Loan(
            loan_id=cursor.lastrowid,
            book_isbn=book_isbn,
            member_id=member_id,
            due_date=due_date,
            returned_date=None,
        )

    def return_loan(self, loan_id: int, return_date: Optional[datetime.date] = None) -> Loan:
        """
        Record the return of a loan.

        Parameters
        ----------
        loan_id     : int
        return_date : datetime.date | None – defaults to today

        Raises
        ------
        ValueError – loan not found or already returned.
        """
        row = self._conn.execute(
            "SELECT * FROM loans WHERE loan_id = ?", (loan_id,)
        ).fetchone()
        if row is None:
            raise ValueError(f"Loan with ID {loan_id} not found")
        if row["returned_date"] is not None:
            raise ValueError(f"Loan {loan_id} has already been returned")

        actual_date = return_date or datetime.date.today()

        # Mark book as available again
        self._conn.execute(
            "UPDATE books SET available = 1 WHERE isbn = ?", (row["book_isbn"],)
        )

        # Record the return date
        self._conn.execute(
            "UPDATE loans SET returned_date = ? WHERE loan_id = ?",
            (actual_date.isoformat(), loan_id),
        )
        self._conn.commit()

        return Loan(
            loan_id=loan_id,
            book_isbn=row["book_isbn"],
            member_id=row["member_id"],
            due_date=datetime.date.fromisoformat(row["due_date"]),
            returned_date=actual_date,
        )

    def get_loan(self, loan_id: int) -> Optional[Loan]:
        """Fetch a single loan by ID.  Returns None if not found."""
        row = self._conn.execute(
            "SELECT * FROM loans WHERE loan_id = ?", (loan_id,)
        ).fetchone()
        if row is None:
            return None
        return self._row_to_loan(row)

    def list_loans(self) -> List[Loan]:
        """Return every loan record, ordered by loan_id."""
        rows = self._conn.execute(
            "SELECT * FROM loans ORDER BY loan_id"
        ).fetchall()
        return [self._row_to_loan(r) for r in rows]

    def list_active_loans(self) -> List[Loan]:
        """Return every loan that has not yet been returned."""
        rows = self._conn.execute(
            "SELECT * FROM loans WHERE returned_date IS NULL ORDER BY due_date"
        ).fetchall()
        return [self._row_to_loan(r) for r in rows]

    def list_overdue_loans(self, as_of: Optional[datetime.date] = None) -> List[Loan]:
        """Return every active loan whose due date has passed."""
        reference = as_of or datetime.date.today()
        rows = self._conn.execute(
            "SELECT * FROM loans WHERE returned_date IS NULL AND due_date < ? ORDER BY due_date",
            (reference.isoformat(),),
        ).fetchall()
        return [self._row_to_loan(r) for r in rows]

    # ── Private helpers ─────────────────────────────────────────
    @staticmethod
    def _row_to_book(row: sqlite3.Row) -> Book:
        return Book(
            isbn=row["isbn"],
            title=row["title"],
            author=row["author"],
            year=row["year"],
            available=bool(row["available"]),
        )

    @staticmethod
    def _row_to_member(row: sqlite3.Row) -> Member:
        return Member(
            member_id=row["member_id"],
            name=row["name"],
            email=row["email"],
        )

    @staticmethod
    def _row_to_loan(row: sqlite3.Row) -> Loan:
        due = datetime.date.fromisoformat(row["due_date"])
        ret = None
        if row["returned_date"] is not None:
            ret = datetime.date.fromisoformat(row["returned_date"])
        return Loan(
            loan_id=row["loan_id"],
            book_isbn=row["book_isbn"],
            member_id=row["member_id"],
            due_date=due,
            returned_date=ret,
        )
```

## File 3: `cli.py` — Command-Line Interface

```python
"""
cli.py
------
Interactive command-line interface for the library management system.

Commands
--------
add-book        Add a new book to the catalogue.
list-books      Show all books.
search-books    Search books by title or author.
remove-book     Delete a book by ISBN.
add-member      Register a new member.
list-members    Show all members.
remove-member   Delete a member by ID.
checkout        Borrow a book (create a loan).
return-book     Return a checked-out book.
list-loans      Show all loans.
list-active     Show outstanding (unreturned) loans.
list-overdue    Show overdue loans.
help            Show available commands.
quit            Exit the application.

Input validation is handled by parsing helpers that raise ValueError with
human-friendly messages; the main loop catches those and re-prompts.
"""

from __future__ import annotations

import datetime
import sys
from typing import Optional

from models import Book
from store import LibraryStore


# ── Prompt helpers with validation ──────────────────────────────

def _prompt_str(label: str) -> str:
    """Prompt for a non-empty string."""
    while True:
        value = input(f"  {label}: ").strip()
        if value:
            return value
        print("  (value cannot be empty)")


def _prompt_int(label: str) -> int:
    """Prompt for a positive integer."""
    while True:
        raw = input(f"  {label}: ").strip()
        try:
            value = int(raw)
            if value <= 0:
                raise ValueError
            return value
        except ValueError:
            print("  (please enter a positive integer)")


def _prompt_year(label: str) -> int:
    """Prompt for a valid publication year."""
    current_year = datetime.date.today().year
    while True:
        raw = input(f"  {label}: ").strip()
        try:
            value = int(raw)
            if value < 1000 or value > current_year:
                raise ValueError
            return value
        except ValueError:
            print(f"  (year must be between 1000 and {current_year})")


def _prompt_date(label: str) -> datetime.date:
    """Prompt for a date in YYYY-MM-DD format."""
    while True:
        raw = input(f"  {label} (YYYY-MM-DD): ").strip()
        try:
            return datetime.date.fromisoformat(raw)
        except ValueError:
            print("  (invalid date – use YYYY-MM-DD)")


def _prompt_isbn(label: str) -> str:
    """Prompt for a valid ISBN (10 or 13 digits, dashes optional)."""
    while True:
        raw = input(f"  {label}: ").strip()
        clean = raw.replace("-", "")
        if clean.isdigit() and len(clean) in (10, 13):
            return raw  # keep original formatting for storage
        print("  (ISBN must be 10 or 13 digits)")


def _prompt_email(label: str) -> str:
    """Prompt for a basic email address."""
    while True:
        raw = input(f"  {label}: ").strip()
        if "@" in raw and "." in raw:
            return raw
        print("  (enter a valid email, e.g. name@example.com)")


# ── Command handlers ────────────────────────────────────────────

def cmd_add_book(store: LibraryStore) -> None:
    """Add a new book to the catalogue."""
    print("\n--- Add Book ---")
    isbn = _prompt_isbn("ISBN")
    title = _prompt_str("Title")
    author = _prompt_str("Author")
    year = _prompt_year("Year")

    try:
        book = Book(isbn=isbn, title=title, author=author, year=year)
        store.add_book(book)
        print(f"  ✓ Added: {book}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_list_books(store: LibraryStore) -> None:
    """Display every book in the catalogue."""
    books = store.list_books()
    if not books:
        print("\n  (catalogue is empty)")
        return
    print(f"\n  {'ISBN':<15} {'Title':<40} {'Author':<25} {'Year':>5}  {'Status'}")
    print("  " + "-" * 95)
    for b in books:
        status = "Available" if b.available else "Checked out"
        print(f"  {b.isbn:<15} {b.title:<40} {b.author:<25} {b.year:>5}  {status}")
    print(f"  Total: {len(books)} book(s)")


def cmd_search_books(store: LibraryStore) -> None:
    """Search the catalogue by title or author."""
    print("\n--- Search Books ---")
    query = _prompt_str("Search term")
    results = store.search_books(query)
    if not results:
        print(f"  No books found matching {query!r}")
        return
    print(f"\n  Found {len(results)} result(s):")
    for b in results:
        status = "Available" if b.available else "Checked out"
        print(f"    {b.isbn}  {b.title}  by {b.author}  ({b.year})  [{status}]")


def cmd_remove_book(store: LibraryStore) -> None:
    """Delete a book by ISBN."""
    print("\n--- Remove Book ---")
    isbn = _prompt_isbn("ISBN")
    book = store.get_book(isbn)
    if book is None:
        print(f"  ✗ Book not found: {isbn}")
        return
    confirm = input(f"  Confirm delete {book.title!r}? (y/n): ").strip().lower()
    if confirm != "y":
        print("  Cancelled.")
        return
    try:
        store.remove_book(isbn)
        print(f"  ✓ Removed: {book.title}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_add_member(store: LibraryStore) -> None:
    """Register a new member."""
    print("\n--- Add Member ---")
    name = _prompt_str("Name")
    email = _prompt_email("Email")
    try:
        member = store.add_member(name=name, email=email)
        print(f"  ✓ Registered: {member}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_list_members(store: LibraryStore) -> None:
    """Display every member."""
    members = store.list_members()
    if not members:
        print("\n  (no members registered)")
        return
    print(f"\n  {'ID':>4}  {'Name':<30}  {'Email'}")
    print("  " + "-" * 70)
    for m in members:
        print(f"  {m.member_id:>4}  {m.name:<30}  {m.email}")
    print(f"  Total: {len(members)} member(s)")


def cmd_remove_member(store: LibraryStore) -> None:
    """Delete a member by ID."""
    print("\n--- Remove Member ---")
    member_id = _prompt_int("Member ID")
    member = store.get_member(member_id)
    if member is None:
        print(f"  ✗ Member not found: {member_id}")
        return
    confirm = input(f"  Confirm delete {member.name!r}? (y/n): ").strip().lower()
    if confirm != "y":
        print("  Cancelled.")
        return
    try:
        store.remove_member(member_id)
        print(f"  ✓ Removed: {member.name}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_checkout(store: LibraryStore) -> None:
    """Check out a book to a member."""
    print("\n--- Checkout ---")
    isbn = _prompt_isbn("Book ISBN")
    member_id = _prompt_int("Member ID")
    due = _prompt_date("Due date")

    try:
        loan = store.create_loan(book_isbn=isbn, member_id=member_id, due_date=due)
        print(f"  ✓ Checked out: {loan}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_return_book(store: LibraryStore) -> None:
    """Return a checked-out book by loan ID."""
    print("\n--- Return Book ---")
    loan_id = _prompt_int("Loan ID")
    try:
        loan = store.return_loan(loan_id)
        print(f"  ✓ Returned: {loan}")
    except Exception as exc:
        print(f"  ✗ Error: {exc}")


def cmd_list_loans(store: LibraryStore) -> None:
    """Display all loans."""
    loans = store.list_loans()
    if not loans:
        print("\n  (no loans recorded)")
        return
    print(f"\n  {'Loan ID':>7}  {'ISBN':<15}  {'Member':>6}  {'Due Date':<12}  {'Status'}")
    print("  " + "-" * 75)
    for loan in loans:
        status = "Active" if loan.is_active() else "Returned"
        overdue = " [OVERDUE]" if loan.is_overdue() else ""
        print(
            f"  {loan.loan_id:>7}  {loan.book_isbn:<15}  "
            f"{loan.member_id:>6}  {loan.due_date.isoformat():<12}  {status}{overdue}"
        )
    print(f"  Total: {len(loans)} loan(s)")


def cmd_list_active(store: LibraryStore) -> None:
    """Display outstanding loans."""
    loans = store.list_active_loans()
    if not loans:
        print("\n  (no outstanding loans)")
        return

    Main entry point for the interactive CLI.

    Parameters
    ----------
    db_path : str – path to the SQLite database file (default: in-memory).
    """

    print("=" * 60)
    print("  Library Management System")
    print("  Type 'help' for a list of commands, 'quit' to exit.")
```

### Key design decisions

| Concern | Decision |
|---|---|
| **Data integrity** | `@dataclass(frozen=True)` makes model objects immutable; `__post_init__` validates every field at construction time. |
| **SQL safety** | Every query uses `?` parameterised placeholders – no string interpolation. |
| **Foreign keys** | `PRAGMA foreign_keys = ON` enforces referential integrity so you cannot delete a book/member with outstanding loans. |
| **Test isolation** | Each test class gets a fresh `:memory:` SQLite database via `setUp` / `tearDown`. |
| **CLI validation** | `while True` loops in `_prompt_*` helpers re-prompt until valid input is supplied; `EOFError` is caught so piping or Ctrl-D exits cleanly. |
| **Loan lifecycle** | `create_loan` atomically flips the book's `available` flag and inserts the loan row in the same transaction; `return_loan` flips it back. |
