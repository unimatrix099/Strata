```python
"""
Data models for the Library Management System.

This module defines the core data classes:
- Book: Represents a book in the library catalog.
- Member: Represents a library member.
- Loan: Represents a loan transaction linking a member to a book.

Each model includes validation logic to ensure data integrity.
"""

from dataclasses import dataclass, field
from datetime import datetime, date
from typing import Optional
import re


# ---------------------------------------------------------------------------
# Validation helpers
# ---------------------------------------------------------------------------

def validate_isbn(isbn: str) -> str:
    """
    Validate an ISBN string.

    Accepts ISBN-10 or ISBN-13 formats.
    - ISBN-10: 10 digits (last digit may be 'X' for 10).
    - ISBN-13: 13 digits, optionally with hyphens.

    Args:
        isbn: The ISBN string to validate.

    Returns:
        The cleaned ISBN string (hyphens removed).

    Raises:
        ValueError: If the ISBN is invalid.
    """
    if not isbn or not isinstance(isbn, str):
        raise ValueError("ISBN must be a non-empty string.")

    # Remove hyphens and spaces for validation
    cleaned = isbn.replace("-", "").replace(" ", "")

    if len(cleaned) == 10:
        # ISBN-10: first 9 digits, last can be digit or 'X'
        if not re.match(r'^\d{9}[\dXx]$', cleaned):
            raise ValueError(
                f"Invalid ISBN-10: '{isbn}'. Must be 9 digits followed by a digit or 'X'."
            )
        return cleaned
    elif len(cleaned) == 13:
        # ISBN-13: all 13 digits
        if not re.match(r'^\d{13}$', cleaned):
            raise ValueError(
                f"Invalid ISBN-13: '{isbn}'. Must be exactly 13 digits."
            )
        return cleaned
    else:
        raise ValueError(
            f"Invalid ISBN: '{isbn}'. Must be 10 or 13 digits (hyphens allowed)."
        )


def validate_title(title: str) -> str:
    """
    Validate a book title.

    Args:
        title: The title string to validate.

    Returns:
        The stripped title string.

    Raises:
        ValueError: If the title is empty or too short.
    """
    if not title or not isinstance(title, str):
        raise ValueError("Title must be a non-empty string.")
    stripped = title.strip()
    if len(stripped) < 1:
        raise ValueError("Title must contain at least one character.")
    if len(stripped) > 500:
        raise ValueError("Title must not exceed 500 characters.")
    return stripped


def validate_author(author: str) -> str:
    """
    Validate an author name.

    Args:
        author: The author name string to validate.

    Returns:
        The stripped author name.

    Raises:
        ValueError: If the author name is empty or too short.
    """
    if not author or not isinstance(author, str):
        raise ValueError("Author must be a non-empty string.")
    stripped = author.strip()
    if len(stripped) < 1:
        raise ValueError("Author name must contain at least one character.")
    if len(stripped) > 200:
        raise ValueError("Author name must not exceed 200 characters.")
    return stripped


def validate_member_name(name: str) -> str:
    """
    Validate a member's name.

    Args:
        name: The member name string to validate.

    Returns:
        The stripped name.

    Raises:
        ValueError: If the name is empty or too short.
    """
    if not name or not isinstance(name, str):
        raise ValueError("Member name must be a non-empty string.")
    stripped = name.strip()
    if len(stripped) < 1:
        raise ValueError("Member name must contain at least one character.")
    if len(stripped) > 100:
        raise ValueError("Member name must not exceed 100 characters.")
    return stripped


def validate_email(email: str) -> str:
    """
    Validate an email address.

    Args:
        email: The email string to validate.

    Returns:
        The stripped email address.

    Raises:
        ValueError: If the email format is invalid.
    """
    if not email or not isinstance(email, str):
        raise ValueError("Email must be a non-empty string.")
    stripped = email.strip()
    # Basic email regex: something@something.something
    if not re.match(r'^[^@\s]+@[^@\s]+\.[^@\s]+$', stripped):
        raise ValueError(f"Invalid email format: '{email}'.")
    return stripped


def validate_loan_date(d: Optional[date]) -> Optional[date]:
    """
    Validate a loan date.

    Args:
        d: The date to validate, or None.

    Returns:
        The validated date or None.

    Raises:
        ValueError: If the date is in the future.
    """
    if d is None:
        return None
    if not isinstance(d, date):
        raise ValueError("Loan date must be a date object.")
    if d > date.today():
        raise ValueError("Loan date cannot be in the future.")
    return d


# ---------------------------------------------------------------------------
# Data classes
# ---------------------------------------------------------------------------

@dataclass
class Book:
    """
    Represents a book in the library catalog.

    Attributes:
        id: Unique identifier (auto-assigned by the database).
        isbn: International Standard Book Number (validated).
        title: Book title (validated).
        author: Author name (validated).
        year: Publication year (optional, 1000–2100).
        available: Whether the book is currently available for loan.
    """
    id: Optional[int] = None
    isbn: str = ""
    title: str = ""
    author: str = ""
    year: Optional[int] = None
    available: bool = True

    def __post_init__(self):
        """Validate fields after initialization."""
        if self.isbn:
            self.isbn = validate_isbn(self.isbn)
        if self.title:
            self.title = validate_title(self.title)
        if self.author:
            self.author = validate_author(self.author)
        if self.year is not None:
            if not isinstance(self.year, int):
                raise ValueError("Year must be an integer.")
            if self.year < 1000 or self.year > 2100:
                raise ValueError("Year must be between 1000 and 2100.")

    def __str__(self) -> str:
        """Return a human-readable string representation."""
        year_str = f" ({self.year})" if self.year else ""
        status = "Available" if self.available else "Checked Out"
        return (
            f"[{self.id}] \"{self.title}\" by {self.author}{year_str} "
            f"[ISBN: {self.isbn}] [{status}]"
        )


@dataclass
class Member:
    """
    Represents a library member.

    Attributes:
        id: Unique identifier (auto-assigned by the database).
        name: Member's full name (validated).
        email: Member's email address (validated).
        joined_date: Date the member joined the library.
    """
    id: Optional[int] = None
    name: str = ""
    email: str = ""
    joined_date: Optional[date] = None

    def __post_init__(self):
        """Validate fields after initialization."""
        if self.name:
            self.name = validate_member_name(self.name)
        if self.email:
            self.email = validate_email(self.email)
        if self.joined_date is not None:
            if not isinstance(self.joined_date, date):
                raise ValueError("Joined date must be a date object.")
            if self.joined_date > date.today():
                raise ValueError("Joined date cannot be in the future.")

    def __str__(self) -> str:
        """Return a human-readable string representation."""
        joined = self.joined_date.isoformat() if self.joined_date else "N/A"
        return f"[{self.id}] {self.name} <{self.email}> (joined: {joined})"


@dataclass
class Loan:
    """
    Represents a loan transaction linking a member to a book.

    Attributes:
        id: Unique identifier (auto-assigned by the database).
        book_id: ID of the borrowed book.
        member_id: ID of the borrowing member.
        loan_date: Date the book was borrowed.
        due_date: Date the book is due back.
        return_date: Date the book was returned (None if not yet returned).
    """
    id: Optional[int] = None
    book_id: int = 0
    member_id: int = 0
    loan_date: Optional[date] = None
    due_date: Optional[date] = None
    return_date: Optional[date] = None

    def __post_init__(self):
        """Validate fields after initialization."""
        if self.book_id <= 0:
            raise ValueError("Book ID must be a positive integer.")
        if self.member_id <= 0:
            raise ValueError("Member ID must be a positive integer.")
        if self.loan_date is not None:
            self.loan_date = validate_loan_date(self.loan_date)
        if self.due_date is not None:
            if not isinstance(self.due_date, date):
                raise ValueError("Due date must be a date object.")
            if self.loan_date and self.due_date < self.loan_date:
                raise ValueError("Due date cannot be before loan date.")
        if self.return_date is not None:
            self.return_date = validate_loan_date(self.return_date)
            if self.loan_date and self.return_date < self.loan_date:
                raise ValueError("Return date cannot be before loan date.")

    def is_overdue(self) -> bool:
        """Check if the loan is overdue (not returned and past due date)."""
        if self.return_date is not None:
            return False
        if self.due_date is None:
            return False
        return date.today() > self.due_date

    def __str__(self) -> str:
        """Return a human-readable string representation."""
        loan_str = self.loan_date.isoformat() if self.loan_date else "N/A"
        due_str = self.due_date.isoformat() if self.due_date else "N/A"
        ret_str = self.return_date.isoformat() if self.return_date else "Not returned"
        overdue_flag = " [OVERDUE]" if self.is_overdue() else ""
        return (
            f"[{self.id}] Book#{self.book_id} -> Member#{self.member_id} "
            f"(loaned: {loan_str}, due: {due_str}, returned: {ret_str}){overdue_flag}"
        )
```

## File 3: `library/store.py`

```python
"""
SQLite-backed store for the Library Management System.

This module provides a Store class that handles all database operations:
- Creating tables
- Adding, retrieving, updating, and deleting books, members, and loans
- Searching and listing records
- Managing loan transactions (checkout, return)

The store uses parameterized queries to prevent SQL injection.
"""

import sqlite3
from datetime import date, timedelta
from typing import List, Optional, Tuple
from pathlib import Path

from library.models import Book, Member, Loan


class Store:
    """
    SQLite-backed store for library data.

    Manages a SQLite database file and provides CRUD operations
    for books, members, and loans.
    """

    def __init__(self, db_path: str = "library.db"):
        """
        Initialize the store with a SQLite database.

        Args:
            db_path: Path to the SQLite database file.
                     Use ":memory:" for an in-memory database (useful for tests).
        """
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path)
        self.conn.row_factory = sqlite3.Row  # Enable column access by name
        self.conn.execute("PRAGMA foreign_keys = ON")  # Enforce foreign keys
        self._create_tables()

    def _create_tables(self) -> None:
        """Create the database tables if they do not already exist."""
        cursor = self.conn.cursor()

        # Books table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS books (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                isbn TEXT UNIQUE NOT NULL,
                title TEXT NOT NULL,
                author TEXT NOT NULL,
                year INTEGER,
                available INTEGER NOT NULL DEFAULT 1
            )
        """)

        # Members table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS members (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                email TEXT UNIQUE NOT NULL,
                joined_date TEXT NOT NULL
            )
        """)

        # Loans table
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS loans (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                book_id INTEGER NOT NULL,
                member_id INTEGER NOT NULL,
                loan_date TEXT NOT NULL,
                due_date TEXT NOT NULL,
                return_date TEXT,
                FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE,
                FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE
            )
        """)

        self.conn.commit()

    # -------------------------------------------------------------------
    # Book operations
    # -------------------------------------------------------------------

    def add_book(self, book: Book) -> Book:
        """
        Add a new book to the catalog.

        Args:
            book: A Book instance (id should be None for new books).

        Returns:
            The Book instance with its assigned id.

        Raises:
            sqlite3.IntegrityError: If the ISBN already exists.
        """
        cursor = self.conn.cursor()
        cursor.execute(
            """
            INSERT INTO books (isbn, title, author, year, available)
            VALUES (?, ?, ?, ?, ?)
            """,
            (book.isbn, book.title, book.author, book.year, int(book.available))
        )
        self.conn.commit()
        book.id = cursor.lastrowid
        return book

    def get_book_by_id(self, book_id: int) -> Optional[Book]:
        """
        Retrieve a book by its ID.

        Args:
            book_id: The book's unique identifier.

        Returns:
            A Book instance, or None if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM books WHERE id = ?", (book_id,))
        row = cursor.fetchone()
        if row is None:
            return None
        return self._row_to_book(row)

    def get_book_by_isbn(self, isbn: str) -> Optional[Book]:
        """
        Retrieve a book by its ISBN.

        Args:
            isbn: The book's ISBN.

        Returns:
            A Book instance, or None if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM books WHERE isbn = ?", (isbn,))
        row = cursor.fetchone()
        if row is None:
            return None
        return self._row_to_book(row)

    def list_books(self, search: Optional[str] = None) -> List[Book]:
        """
        List all books, optionally filtered by a search term.

        Args:
            search: Optional search string to match against title, author, or ISBN.

        Returns:
            A list of Book instances.
        """
        cursor = self.conn.cursor()
        if search:
            like_pattern = f"%{search}%"
            cursor.execute(
                """
                SELECT * FROM books
                WHERE title LIKE ? OR author LIKE ? OR isbn LIKE ?
                ORDER BY title
                """,
                (like_pattern, like_pattern, like_pattern)
            )
        else:
            cursor.execute("SELECT * FROM books ORDER BY title")
        rows = cursor.fetchall()
        return [self._row_to_book(row) for row in rows]

    def update_book(self, book: Book) -> Book:
        """
        Update an existing book in the catalog.

        Args:
            book: A Book instance with an existing id.

        Returns:
            The updated Book instance.

        Raises:
            ValueError: If the book id is None or the book does not exist.
        """
        if book.id is None:
            raise ValueError("Cannot update a book without an id.")

        cursor = self.conn.cursor()
        cursor.execute(
            """
            UPDATE books
            SET isbn = ?, title = ?, author = ?, year = ?, available = ?
            WHERE id = ?
            """,
            (book.isbn, book.title, book.author, book.year, int(book.available), book.id)
        )
        self.conn.commit()
        if cursor.rowcount == 0:
            raise ValueError(f"Book with id {book.id} not found.")
        return book

    def delete_book(self, book_id: int) -> bool:
        """
        Delete a book from the catalog.

        Args:
            book_id: The book's unique identifier.

        Returns:
            True if the book was deleted, False if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("DELETE FROM books WHERE id = ?", (book_id,))
        self.conn.commit()
        return cursor.rowcount > 0

    # -------------------------------------------------------------------
    # Member operations
    # -------------------------------------------------------------------

    def add_member(self, member: Member) -> Member:
        """
        Add a new member to the library.

        Args:
            member: A Member instance (id should be None for new members).

        Returns:
            The Member instance with its assigned id.

        Raises:
            sqlite3.IntegrityError: If the email already exists.
        """
        cursor = self.conn.cursor()
        joined = member.joined_date or date.today()
        cursor.execute(
            """
            INSERT INTO members (name, email, joined_date)
            VALUES (?, ?, ?)
            """,
            (member.name, member.email, joined.isoformat())
        )
        self.conn.commit()
        member.id = cursor.lastrowid
        member.joined_date = joined
        return member

    def get_member_by_id(self, member_id: int) -> Optional[Member]:
        """
        Retrieve a member by their ID.

        Args:
            member_id: The member's unique identifier.

        Returns:
            A Member instance, or None if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM members WHERE id = ?", (member_id,))
        row = cursor.fetchone()
        if row is None:
            return None
        return self._row_to_member(row)

    def get_member_by_email(self, email: str) -> Optional[Member]:
        """
        Retrieve a member by their email.

        Args:
            email: The member's email address.

        Returns:
            A Member instance, or None if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM members WHERE email = ?", (email,))
        row = cursor.fetchone()
        if row is None:
            return None
        return self._row_to_member(row)

    def list_members(self, search: Optional[str] = None) -> List[Member]:
        """
        List all members, optionally filtered by a search term.

        Args:
            search: Optional search string to match against name or email.

        Returns:
            A list of Member instances.
        """
        cursor = self.conn.cursor()
        if search:
            like_pattern = f"%{search}%"
            cursor.execute(
                """
                SELECT * FROM members
                WHERE name LIKE ? OR email LIKE ?
                ORDER BY name
                """,
                (like_pattern, like_pattern)
            )
        else:
            cursor.execute("SELECT * FROM members ORDER BY name")
        rows = cursor.fetchall()
        return [self._row_to_member(row) for row in rows]

    def update_member(self, member: Member) -> Member:
        """
        Update an existing member.

        Args:
            member: A Member instance with an existing id.

        Returns:
            The updated Member instance.

        Raises:
            ValueError: If the member id is None or the member does not exist.
        """
        if member.id is None:
            raise ValueError("Cannot update a member without an id.")

        cursor = self.conn.cursor()
        joined = member.joined_date or date.today()
        cursor.execute(
            """
            UPDATE members
            SET name = ?, email = ?, joined_date = ?
            WHERE id = ?
            """,
            (member.name, member.email, joined.isoformat(), member.id)
        )
        self.conn.commit()
        if cursor.rowcount == 0:
            raise ValueError(f"Member with id {member.id} not found.")
        return member

    def delete_member(self, member_id: int) -> bool:
        """
        Delete a member from the library.

        Args:
            member_id: The member's unique identifier.

        Returns:
            True if the member was deleted, False if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("DELETE FROM members WHERE id = ?", (member_id,))
        self.conn.commit()
        return cursor.rowcount > 0

    # -------------------------------------------------------------------
    # Loan operations
    # -------------------------------------------------------------------

    def checkout_book(
        self,
        book_id: int,
        member_id: int,
        loan_days: int = 14
    ) -> Loan:
        """
        Check out a book to a member.

        Args:
            book_id: The ID of the book to borrow.
            member_id: The ID of the borrowing member.
            loan_days: Number of days for the loan (default: 14).

        Returns:
            A Loan instance representing the transaction.

        Raises:
            ValueError: If the book is not found, not available, or the member
                        is not found.
        """
        # Verify book exists and is available
        book = self.get_book_by_id(book_id)
        if book is None:
            raise ValueError(f"Book with id {book_id} not found.")
        if not book.available:
            raise ValueError(f"Book '{book.title}' (id {book_id}) is not available.")

        # Verify member exists
        member = self.get_member_by_id(member_id)
        if member is None:
            raise ValueError(f"Member with id {member_id} not found.")

        # Create the loan
        loan_date = date.today()
        due_date = loan_date + timedelta(days=loan_days)

        cursor = self.conn.cursor()
        cursor.execute(
            """
            INSERT INTO loans (book_id, member_id, loan_date, due_date)
            VALUES (?, ?, ?, ?)
            """,
            (book_id, member_id, loan_date.isoformat(), due_date.isoformat())
        )
        self.conn.commit()
        loan_id = cursor.lastrowid

        # Mark the book as unavailable
        cursor.execute(
            "UPDATE books SET available = 0 WHERE id = ?",
            (book_id,)
        )
        self.conn.commit()

        return Loan(
            id=loan_id,
            book_id=book_id,
            member_id=member_id,
            loan_date=loan_date,
            due_date=due_date,
            return_date=None
        )

    def return_book(self, loan_id: int) -> Loan:
        """
        Return a borrowed book.

        Args:
            loan_id: The ID of the loan to close.

        Returns:
            The updated Loan instance with return_date set.

        Raises:
            ValueError: If the loan is not found or already returned.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM loans WHERE id = ?", (loan_id,))
        row = cursor.fetchone()
        if row is None:
            raise ValueError(f"Loan with id {loan_id} not found.")

        loan = self._row_to_loan(row)
        if loan.return_date is not None:
            raise ValueError(f"Loan {loan_id} has already been returned.")

        return_date = date.today()
        cursor.execute(
            "UPDATE loans SET return_date = ? WHERE id = ?",
            (return_date.isoformat(), loan_id)
        )
        self.conn.commit()

        # Mark the book as available again
        cursor.execute(
            "UPDATE books SET available = 1 WHERE id = ?",
            (loan.book_id,)
        )
        self.conn.commit()

        loan.return_date = return_date
        return loan

    def get_loan_by_id(self, loan_id: int) -> Optional[Loan]:
        """
        Retrieve a loan by its ID.

        Args:
            loan_id: The loan's unique identifier.

        Returns:
            A Loan instance, or None if not found.
        """
        cursor = self.conn.cursor()
        cursor.execute("SELECT * FROM loans WHERE id = ?", (loan_id,))
        row = cursor.fetchone()
        if row is None:
            return None
        return self._row_to_loan(row)

    def list_loans(
        self,
        member_id: Optional[int] = None,
        book_id: Optional[int] = None,
        active_only: bool = False
    ) -> List[Loan]:
        """
        List loans, optionally filtered by member, book, or active status.

        Args:
            member_id: Filter by member ID (optional).
            book_id: Filter by book ID (optional).
            active_only: If True, only return loans that have not been returned.

        Returns:
            A list of Loan instances.
        """
        cursor = self.conn.cursor()
        query = "SELECT * FROM loans WHERE 1=1"
        params: List = []

        if member_id is not None:
            query += " AND member_id = ?"
            params.append(member_id)
        if book_id is not None:
            query += " AND book_id = ?"
            params.append(book_id)
        if active_only:
            query += " AND return_date IS NULL"

        query += " ORDER BY loan_date DESC"
        cursor.execute(query, params)
        rows = cursor.fetchall()
        return [self._row_to_loan(row) for row in rows]

    def get_overdue_loans(self) -> List[Loan]:
        """
        Get all loans that are overdue (not returned and past due date).

        Returns:
            A list of overdue Loan instances.
        """
        today = date.today().isoformat()
        cursor = self.conn.cursor()
        cursor.execute(
            """
            SELECT * FROM loans
            WHERE return_date IS NULL AND due_date < ?
            ORDER BY due_date
            """,
            (today,)
        )
        rows = cursor.fetchall()
        return [self._row_to_loan(row) for row in rows]

    # -------------------------------------------------------------------
    # Row-to-model conversion helpers
    # -------------------------------------------------------------------

    @staticmethod
    def _row_to_book(row: sqlite3.Row) -> Book:
        """Convert a database row to a Book instance."""
        return Book(
            id=row["id"],
            isbn=row["isbn"],
            title=row["title"],
            author=row["author"],
            year=row["year"],
            available=bool(row["available"])
        )

    @staticmethod
    def _row_to_member(row: sqlite3.Row) -> Member:
        """Convert a database row to a Member instance."""
        joined = None
        if row["joined_date"]:
            joined = date.fromisoformat(row["joined_date"])
        return Member(
            id=row["id"],
            name=row["name"],
            email=row["email"],
            joined_date=joined
        )

    @staticmethod
    def _row_to_loan(row: sqlite3.Row) -> Loan:
        """Convert a database row to a Loan instance."""
        loan_date = None
        due_date = None
        return_date = None
        if row["loan_date"]:
            loan_date = date.fromisoformat(row["loan_date"])
        if row["due_date"]:
```
